\begin{abstract}
% PLACEHOLDER --- REWRITE
% Tightened from docs/novelty-sweep.md \S5's recommended positioning
% paragraph to <=600 characters (PRL's hard abstract limit). Rendered
% plain-text length of the paragraph below (LaTeX markup stripped, math
% read as plain symbols) measured at 573 characters -- verify again after
% any edit; PRL will bounce the submission over 600.
We prove that the infrared triangle of continuum field theory --
soft theorem, asymptotic symmetry, memory effect -- closes exactly,
at finite lattice spacing, in a 1D quantum spin chain. Using the
matrix-product-state excitation ansatz we derive an Adler-zero-type
soft theorem for magnons of the Heisenberg ferromagnet from symmetry
intertwiners, realize $(G_L\times G_R)/G_{\mathrm{diag}}$ as virtual
bond operators whose projective class is an SPT index, and obtain the
magnon memory effect as a Ward identity. The three corners are
provably equivalent, with no continuum limit invoked.
\end{abstract}

\textit{Lead-in.---} [DRAFT -- narrative beat (iv): state the soft

\begin{theorem}[Lattice soft magnon theorem -- DRAFT STATEMENT]
% PROVED / CONJECTURE status to be stamped in per claims/CLAIMS.md claim
% G0 before this is submitted; do not weaken or strengthen the hedge
% without an L6 critic pass recorded in theory/corner-a-goldstone.md.
Let $\xi$ generate a (possibly spontaneously broken) one-parameter
subgroup of $G$ realized on the uMPS ground state $\ket{\Psi(A)}$, with
Noether current mode $J_k(\xi)$ and charge mode $Q_k(\xi)$. Then
\[
  [H,\,Q_k(\xi)] \;=\; (e^{\ic k}-1)\,J_k(\xi),
\]
with prefactor $(e^{\ic k}-1)$ independent of $H$, of $\xi$, and of the
choice of ground state in the folium. \emph{[STATEMENT DRAFT -- verify
against theory/corner-a-goldstone.md claim G0 verbatim before use; this
is a paraphrase, not a copy, and may have dropped a hypothesis.]}
\end{theorem}
